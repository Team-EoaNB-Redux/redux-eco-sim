test_if_else_chain = {
	set_temp_variable = { x = 5 }
	if = { limit = { check_variable = { x > 10 } } log = "big" }
	else_if = { limit = { check_variable = { x > 3 } } log = "medium" }
	else = { log = "small" }
	if = { limit = { check_variable = { x < 3 } } log = "tiny" }
	else = { log = "not tiny" }
	# expect: medium
	# expect: not tiny
}

test_check_variable_forms = {
	set_temp_variable = { x = 5 }
	if = { limit = { check_variable = { x = 5 } } log = "eq" }
	if = { limit = { check_variable = { x = 4 } } log = "eq is not >=" }
	if = { limit = { check_variable = { var = x value = 5 compare = greater_than_or_equals } } log = "long ge" }
	if = { limit = { check_variable = { var = x value = 4 } } log = "long default ge" }
	if = { limit = { check_variable = { x > 5 } } log = "strict" }
	# expect: eq
	# expect: long ge
	# expect: long default ge
}

test_not_is_nor = {
	if = { limit = { NOT = { always = no always = no } } log = "nor of falses" }
	if = { limit = { NOT = { always = no always = yes } } log = "wrong" }
	if = { limit = { OR = { always = no always = yes } } log = "or" }
	# expect: nor of falses
	# expect: or
}

test_arrays = {
	add_to_array = { nums = 10 }
	add_to_array = { nums = 20 }
	add_to_array = { array = nums value = 15 index = 1 }
	log = "[?nums^num|0]: [?nums^0|0] [?nums^1|0] [?nums^2|0]"
	remove_from_array = { nums = 15 }
	remove_from_array = { array = nums index = 0 }
	log = "[?nums^num|0]: [?nums^0|0]"
	add_to_temp_array = { tmp = 3 }
	add_to_temp_array = { tmp = 9 }
	add_to_temp_array = { tmp = 1 }
	find_highest_in_array = { array = tmp value = hv index = hi }
	log = "[?hv|0] at [?hi|0]"
	resize_array = { array = nums size = 3 value = 7 }
	log = "[?nums^num|0] [?nums^2|0]"
	if = { limit = { is_in_array = { nums = 7 } } log = "has 7" }
	clear_array = nums
	log = "[?nums^num|0]"
	# expect: 3: 10 15 20
	# expect: 1: 20
	# expect: 9 at 1
	# expect: 3 7
	# expect: has 7
	# expect: 0
}

test_for_each_loop_and_break = {
	add_to_temp_array = { a = 1 }
	add_to_temp_array = { a = 2 }
	add_to_temp_array = { a = 3 }
	add_to_temp_array = { a = 4 }
	set_temp_variable = { sum = 0 }
	for_each_loop = {
		array = a
		value = v
		index = i
		break = bb
		add_to_temp_variable = { sum = v }
		if = { limit = { check_variable = { i = 1 } } set_temp_variable = { bb = 1 } }
		add_to_temp_variable = { sum = 100 }
	}
	log = "[?sum|0]"
	# 1 + 100 (i=0), then 2 (i=1) breaks instantly before +100
	# expect: 103
}

test_meta_effect = {
	set_variable = { steel_foundry_size = 7 }
	set_temp_variable = { industry_token = token:steel_foundry }
	meta_effect = {
		text = {
			if = { limit = { has_variable = [INDUSTRY]_size } set_variable = { size@var:industry_token = [INDUSTRY]_size } }
		}
		INDUSTRY = "[?industry_token.GetTokenKey]"
	}
	log = "[?size@token:steel_foundry|0]"
	set_temp_variable = { lvl = 3.7 }
	meta_effect = {
		text = { set_temp_variable = { out = [N] } }
		N = "[?lvl|0]"
	}
	log = "[?out|0]"
	# expect: 7
	# expect: 3
}
